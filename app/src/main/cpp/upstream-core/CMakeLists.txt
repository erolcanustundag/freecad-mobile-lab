cmake_minimum_required(VERSION 3.18)
project(FreeCADMobileDocumentBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python3 REQUIRED COMPONENTS Development.Embed)
if(ANDROID)
    find_package(Qt6 REQUIRED COMPONENTS Core)
    set(QT_CORE Qt6::Core)
else()
    find_package(Qt5 REQUIRED COMPONENTS Core)
    set(QT_CORE Qt5::Core)
endif()
add_library(freecad_python SHARED IMPORTED)
set_target_properties(freecad_python PROPERTIES IMPORTED_LOCATION "${FREECAD_CORE_LIBRARY}")
add_library(freecad_document_bridge SHARED CoreBridge.cpp)
target_link_libraries(freecad_document_bridge PRIVATE freecad_python Python3::Python ${QT_CORE})
if(FREECAD_PART_LIBRARY AND FREECAD_MATERIALS_LIBRARY)
    add_library(freecad_part SHARED IMPORTED)
    set_target_properties(freecad_part PROPERTIES IMPORTED_LOCATION "${FREECAD_PART_LIBRARY}")
    add_library(freecad_materials SHARED IMPORTED)
    set_target_properties(freecad_materials PROPERTIES IMPORTED_LOCATION "${FREECAD_MATERIALS_LIBRARY}")
    target_link_libraries(freecad_document_bridge PRIVATE freecad_part freecad_materials)
    target_compile_definitions(freecad_document_bridge PRIVATE FREECAD_MOBILE_HAS_PART=1)
endif()
if(FREECAD_SKETCHER_LIBRARY)
    if(NOT FREECAD_PART_LIBRARY OR NOT FREECAD_MATERIALS_LIBRARY)
        message(FATAL_ERROR "Original Sketcher requires original Part and Materials")
    endif()
    add_library(freecad_original_sketcher SHARED IMPORTED)
    set_target_properties(freecad_original_sketcher PROPERTIES IMPORTED_LOCATION "${FREECAD_SKETCHER_LIBRARY}")
    target_link_libraries(freecad_document_bridge PRIVATE freecad_original_sketcher)
    target_compile_definitions(freecad_document_bridge PRIVATE FREECAD_MOBILE_HAS_SKETCHER=1)
endif()
if(ANDROID)
    target_link_options(freecad_document_bridge PRIVATE -Wl,-z,max-page-size=16384)
else()
    target_include_directories(freecad_document_bridge PRIVATE "${JDK_ROOT}/include" "${JDK_ROOT}/include/linux")
endif()
